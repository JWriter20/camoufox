# Post-patch semantic assertions.
#
# WHY THIS EXISTS
# ---------------
# `patch` reports success when a hunk's *context* matches — it does NOT verify
# the hunk landed in the function/region you intended. With fuzz (GNU patch
# defaults to fuzz=2) a hunk can silently apply to the WRONG place and still
# exit 0 with no .rej file. This bit us on the FF150 upgrade: the file-picker
# interception block in patches/playwright/0-playwright.patch applied to
# InitColorPicker() instead of InitFilePicker() because v150 gave both
# functions identical surrounding context. It compiled fine, shipped, and
# silently broke every resume upload (filechooser never fired -> fallback to
# setInputFiles -> FlashCollection deterministic-fill tests failed).
#
# These assertions run AFTER all patches apply and BEFORE the build. Any
# unmet assertion aborts patch.py with a non-zero exit, so a misapplied or
# no-op patch fails loudly instead of producing a quietly-broken browser.
#
# FORMAT (one assertion per line; '#' and blank lines ignored)
# Fields are separated by ' ||| ' (space-pipe-pipe-pipe-space) so C++ tokens
# like '::' and '||' can appear freely inside a field.
#
#   contains ||| <file> ||| <substring>
#       <substring> must appear somewhere in <file>.
#
#   absent ||| <file> ||| <substring>
#       <substring> must NOT appear anywhere in <file>.
#
#   in_fn ||| <file> ||| <fn_signature_substring> ||| <substring>
#       <substring> must appear AFTER the line containing <fn_signature_substring>
#       and BEFORE the next top-level function-opening line. Pins a hunk to a
#       specific function — this is the check that catches wrong-function apply.
#
#   absent_in_fn ||| <file> ||| <fn_signature_substring> ||| <substring>
#       <substring> must NOT appear inside the named function's body.
#
# Paths are relative to the Firefox source root (the dir patch.py cd's into),
# AFTER copy-additions.sh has run — so juggler additions live at juggler/...,
# not additions/juggler/...

# --- WebRTC IP spoof must read the spoof IP from config (patches/webrtc-ip-spoofing.patch) ---
# The spoof gate (ShouldSpoofCandidateIP) and the mask source (getMaskForIP)
# and the stats sanitizer MUST read config["webrtc:ipv4"/"ipv6"] via MaskConfig.
# An earlier build wired these to WebRTCIPManager storage that was only ever
# filled by a window.setWebRTCIPv4() JS call that nothing ever made -> the gate
# always returned false, the spoof never ran, and the real public IP leaked in
# every srflx ICE candidate. We assert presence with plain `contains` (the
# in_fn body-bounder can't close these tiny functions: their body lines contain
# `::GetString(`, which trips the fn_open regex). The whole-file `absent` is the
# real regression guard: if the dead WebRTCIPManager read path ever comes back,
# this fails loudly. The three GetString("webrtc:ipv4") sites are gate +
# getMaskForIP + stats sanitizer.

contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| MaskConfig::GetString("webrtc:ipv6")
absent ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| WebRTCIPManager::GetIPv4
absent ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| WebRTCIPManager::GetIPv6

# --- WebRTC synthetic-srflx fabrication (TCP-proxy n/a fix) ---
# Behind a TCP-only proxy STUN (UDP) can't form a srflx, so the rewrite-only
# spoof has nothing to operate on -> WebRTC reports n/a (anti-bot tell). We
# fabricate a synthetic srflx carrying the spoof/proxy IP into the candidate
# stream, the local SDP, AND getStats() (all three must agree or the mismatch
# itself leaks). These assertions fail loudly if any of the three injection
# sites or the gathering-complete trigger is dropped on a future regen.
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| MaybeFabricateSrflxCandidate
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| typ srflx raddr 0.0.0.0 rport 0
# Trigger must live in IceGatheringStateChange (fires the synthetic candidate
# before the null-candidate "done" event at gathering-complete).
in_fn ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| void PeerConnectionImpl::IceGatheringStateChange( ||| MaybeFabricateSrflxCandidate
# getStats() must synthesize a matching srflx local-candidate when the SDP has
# one but the real stats don't (self-consistency across surfaces).
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| RTCIceCandidateType::Srflx
# The fabrication must emit a typ-host candidate (foundation 0) AND a TCP host
# (foundation 2) before the srflx, matching real Firefox's two-candidate shape.
# Without these CreepJS shows the srflx in the host box / foundation 1 (a tell).
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| typ host tcptype active
# The srflx must be DELAYED (realistic STUN round-trip) not emitted simultaneously
# with the host -- a zero-delay srflx is an unnatural timing signature.
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| RandomSrflxDelayMs
# Priorities must be the exact real-Firefox constants (varying them is a tell).
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| 2122252543
contains ||| dom/media/webrtc/jsapi/PeerConnectionImpl.cpp ||| 1686052863

# --- juggler MUST be built + packaged into omni.ja (patches/playwright/0-playwright.patch) ---
# These two hunks are what actually puts the juggler chrome JS into the
# browser. Without them the C++ observer hooks above still compile, the build
# succeeds, and Playwright/camoufox-js silently CANNOT drive the browser
# (no `content juggler` in chrome.manifest -> 0 juggler files in omni.ja).
# This is the "retarget trap": building after `make set-target`
# (patch.py --mozconfig-only) on a tree patched for a DIFFERENT target leaves
# these registration hunks unapplied. Always full-patch per target; these
# assertions fail the build loudly if juggler would be dropped from omni.ja.
# 1) DIRS registration — juggler is added under the ENABLE_WEBDRIVER block.
contains ||| toolkit/toolkit.mozbuild ||| "/juggler"
# 2) Packaging — the juggler jar + manifest must be listed in the installer
#    manifest so they land in omni.ja.
contains ||| browser/installer/package-manifest.in ||| chrome/juggler.manifest
